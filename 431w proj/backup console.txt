CREATE TABLE Address(address_id TEXT PRIMARY KEY,
                     zipcode INTEGER,
                     street_num INTEGER,
                     street_name TEXT,
                     FOREIGN KEY(zipcode) REFERENCES Zipcode(zipcode));

CREATE TABLE Auction_Listing(seller_email TEXT,
                             listing_ID INTEGER PRIMARY KEY,
                             category TEXT,
                             auction_title TEXT,
                             product_name TEXT,
                             product_description TEXT,
                             quantity INTEGER,
                             reserve_price TEXT,
                             max_bids INTEGER,
                             status INTEGER,
                             FOREIGN KEY(seller_email) REFERENCES User(email),
                             FOREIGN KEY(category) REFERENCES Category(category_name));

CREATE TABLE Bidder(email TEXT PRIMARY KEY,
                    first_name TEXT,
                    last_name TEXT,
                    age INTEGER,
                    home_address_id TEXT,
                    major TEXT,
                    FOREIGN KEY(email) REFERENCES User(email),
                    FOREIGN KEY(home_address_id) REFERENCES Address(address_id));

CREATE TABLE Bid(bid_ID INTEGER PRIMARY KEY,
                 seller_email TEXT,
                 listing_ID INTEGER,
                 bidder_email TEXT,
                 bid_price INTEGER,
                 FOREIGN KEY(seller_email) REFERENCES User(email),
                 FOREIGN KEY(listing_ID) REFERENCES Auction_Listing(listing_ID),
                 FOREIGN KEY(bidder_email) REFERENCES User(email));

CREATE TABLE Category(parent_category TEXT,
                      category_name TEXT PRIMARY KEY,
                      FOREIGN KEY(parent_category) REFERENCES Category(category_name));

CREATE TABLE Credit_Card(credit_card_num TEXT PRIMARY KEY,
                         card_type TEXT,
                         expire_month INTEGER,
                         expire_year INTEGER,
                         security_code INTEGER,
                         owner_email TEXT,
                         FOREIGN KEY(owner_email) REFERENCES User(email));

CREATE TABLE Helpdesk(email TEXT PRIMARY KEY,
                      position TEXT,
                      FOREIGN KEY(email) REFERENCES User(email));

CREATE TABLE Local_Vendor(email	TEXT PRIMARY KEY,
                          business_name TEXT,
                          business_address_ID TEXT,
                          customer_service_phone_number TEXT,
                          FOREIGN KEY(email) REFERENCES User(email),
                          FOREIGN KEY(business_address_ID) REFERENCES Address(address_id));

CREATE TABLE Rating(bidder_email TEXT,
                    seller_email TEXT,
                    date DATE,
                    rating INTEGER,
                    rating_desc TEXT,
                    PRIMARY KEY(bidder_email, seller_email),
                    FOREIGN KEY(bidder_email) REFERENCES User(email),
                    FOREIGN KEY(seller_email) REFERENCES User(email));

CREATE TABLE Request(request_id INTEGER PRIMARY KEY,
                     sender_email TEXT,
                     helpdesk_staff_email TEXT,
                     request_type TEXT,
                     request_desc TEXT,
                     request_status INTEGER,
                     FOREIGN KEY(sender_email) REFERENCES User(email),
                     FOREIGN KEY(helpdesk_staff_email) REFERENCES User(email));

CREATE TABLE Seller(email TEXT PRIMARY KEY,
                    bank_routing_number TEXT,
                    bank_account_number INTEGER,
                    balance INTEGER,
                    FOREIGN KEY(email) REFERENCES User(email));

CREATE TABLE Transact(transaction_ID INTEGER PRIMARY KEY,
                      seller_email TEXT,
                      listing_ID INTEGER,
                      bidder_email TEXT,
                      date DATE,
                      payment INTEGER,
                      FOREIGN KEY(seller_email) REFERENCES User(email),
                      FOREIGN KEY(bidder_email) REFERENCES User(email));

CREATE TABLE User(email TEXT PRIMARY KEY,
                  password TEXT);

CREATE TABLE Zipcode(zipcode INTEGER PRIMARY KEY,
                     city TEXT,
                     state TEXT);
